# Example: triangle with altitude, median and circumcircle
# Render: node scripts/render.js --out triangle.png --cmd-file cmds.txt
A=(0.5,0.5)
B=(7.0,0.5)
C=(3.0,5.2)
t1=Polygon(A,B,C)
c1=Circle(A,B)
h=PerpendicularLine(C,Line(A,B))
D=Intersect(h,Line(A,B))
M=Midpoint(A,B)
s1=Segment(D,M)
setColor(t1,"#1c7ed6")
setLineThickness(t1,3)
setColor(c1,"#f08c00")
setLineThickness(c1,2)
setColor(D,"#c92a2a")
setPointSize(D,10)
setColor(M,"#2b8a3e")
setPointSize(M,9)
setColor(s1,"#2b8a3e")
setLineThickness(s1,3)
hide(h)
